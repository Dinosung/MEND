"""Sentiment tagging for customer reviews using Claude."""

MODEL = "claude-sonnet-4-6"
LABELS = ("positive", "negative", "neutral")

SYSTEM_PROMPT = (
    "You classify the sentiment of customer reviews. "
    "Respond with exactly one lower-case word: positive, negative, or neutral. "
    "Do not add punctuation or any other text."
)


def classify_sentiment(client, text):
    """Return "positive", "negative" or "neutral" for the given review text."""
    response = client.messages.create(
        model=MODEL,
        max_tokens=16,
        # Sonnet 4.6 still accepts sampling parameters; temperature 0 makes the
        # label as repeatable as the API allows (not strictly deterministic).
        temperature=0,
        system=SYSTEM_PROMPT,
        messages=[{"role": "user", "content": f"Review:\n{text}"}],
    )

    answer = "".join(
        block.text for block in response.content if block.type == "text"
    )
    label = answer.strip().strip(".!\"'").lower()
    return label if label in LABELS else "neutral"
