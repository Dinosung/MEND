"""Sentiment tagging for customer reviews using Claude."""

from __future__ import annotations

import re

MODEL = "claude-sonnet-4-6"
LABELS = ("positive", "negative", "neutral")

SYSTEM_PROMPT = (
    "You classify the sentiment of customer reviews. "
    "Answer with exactly one word: positive, negative, or neutral. "
    "Do not add punctuation or explanation."
)


def classify_sentiment(client, text):
    """Return "positive", "negative" or "neutral" for a review."""
    response = client.messages.create(
        model=MODEL,
        max_tokens=16,
        # Sonnet 4.6 still accepts sampling params; temperature 0 makes the
        # label as repeatable as the API allows. Thinking is left off (the
        # default when omitted on this model) so there's no extra variance.
        temperature=0,
        system=SYSTEM_PROMPT,
        messages=[{"role": "user", "content": f"Review:\n{text}"}],
    )
    answer = "".join(
        block.text for block in response.content if block.type == "text"
    )
    match = re.search(r"[a-z]+", answer.strip().lower())
    word = match.group(0) if match else ""
    return word if word in LABELS else "neutral"
